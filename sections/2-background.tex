\subsection{Foundations of Measure Theory}
We begin by recalling some basic notions from measure theory, which will be used to formulate the optimal transport problem in a measure-theoretic setting. 
\begin{definition}[$\sigma$-Algebra]
	Let $X$ be a set. A family $\mathcal{A}\subset \mathcal{P}(X)$ is called a \emph{$\sigma$-algebra} on $X$ if%We call $\mathcal{A}$ a \textbf{$\sigma$-algebra} over $X$ if it satisfies
	\begin{enumerate}
		\item $\emptyset \in \mathcal{A}$,
		\item $A\in \mathcal{A} \implies A^C\in \mathcal{A}$,
		\item For every sequence $(A_i)_{i\in \mathbb{N}} \subset \mathcal{A} \implies \bigcup\limits_{i=1}^\infty A_i \in \mathcal{A}.$
	\end{enumerate}
\end{definition}
The pair $(X,\mathcal{A})$ is called a
\emph{measurable space}, and the elements of
$\mathcal{A}$ are called measurable sets.

\begin{definition}[Borel $\sigma$-Algebra]
    Let $X$ be a topological space with topology
    $\mathcal{O}$. The smallest $\sigma$-algebra
    containing all open subsets of $X$ is called
    the \emph{Borel $\sigma$-algebra} and is
    denoted by $\mathcal{B}(X)$.
\end{definition}

\begin{definition}[Measures]
    Let $(X,\mathcal{A})$ be a measurable space.
    A map $\mu:\mathcal{A}\to[0,\infty]$ is called
    a \emph{positive measure} if
    $\mu(\emptyset)=0$ and
    \[
        \mu\left(\bigcup_{i=1}^{\infty}A_i\right)
        =\sum_{i=1}^{\infty}\mu(A_i)
    \]
    for every sequence of pairwise disjoint
    sets $(A_i)_{i\in\mathbb{N}}$ in $\mathcal{A}$.
    This property is called $\sigma$-additivity.

    A measure $\mu$ is called
    \emph{finite} if $\mu(X)<\infty$ and
    \emph{$\sigma$-finite} if there exists a
    sequence $(A_i)_{i\in\mathbb{N}}$ in
    $\mathcal{A}$ such that
    \[
        X=\bigcup_{i=1}^{\infty}A_i,
        \qquad \mu(A_i)<\infty.
    \]
    A positive measure satisfying $\mu(X)=1$
	is called a \emph{probability measure}. A map $\nu:\mathcal{A}\to(-\infty,\infty]$
    is called a \emph{finite signed measure}
    if $\nu(\emptyset)=0$ and it satisfies
    the same $\sigma$-additivity property.
\end{definition}

A measurable set $A$ is called a
\emph{$\mu$-null set} if $\mu(A)=0$.
For a signed measure, null sets will be
understood with respect to its total
variation, defined below. For the remainder of this subsection, let
$X$ be a locally compact Hausdorff space
equipped with its Borel $\sigma$-algebra
$\mathcal{B}(X)$.

\begin{definition}[Radon Measure]
    A positive Borel measure $\mu$ on $X$
    is called a \emph{Radon measure} if
    \begin{enumerate}
        \item $\mu(K)<\infty$ for every
              compact set $K\subseteq X$,
        \item for every $A\in\mathcal{B}(X)$,
              \[
                  \mu(A)
                  =\sup\{\mu(K):
                  K\subseteq A,\;
                  K\text{ compact}\}.
              \]
    \end{enumerate}
    A finite signed Borel measure is called
    a Radon measure if its total variation
    is a Radon measure.
\end{definition}

\begin{definition}[Total Variation]
    Let $\mu$ be a finite signed measure on
    $(X,\mathcal{B}(X))$. Its
    \emph{total variation measure} is defined by
    \[
        |\mu|(A)
        :=\sup\left\{
            \sum_{i=1}^{n}|\mu(A_i)|
            \;\middle|\;
            %\begin{array}{l}
                %n\in\mathbb{N},\\
                A=\bigcup_{i=1}^{n}A_i
                            %\end{array}
        \right\},
    \]
	for $A_i\in\mathcal{B}(X)\text{ pairwise disjoint}$.

    The corresponding \emph{total variation norm}
    is given by
    \[
        \|\mu\|_{\mathrm{TV}}:=|\mu|(X).
    \]
\end{definition}
%TODO Dazu noch mehr ausführen?
The total variation is a positive measure
and quantifies the magnitude of a signed
measure independently of cancellations
between its positive and negative parts.
In particular, for a positive measure $\mu$,
we have $|\mu|=\mu$.

\begin{definition}[Absolute Continuity and Singularity]
    Let $\mu$ and $\nu$ be finite signed
    measures on $(X,\mathcal{B}(X))$. We say that $\nu$ is
    \emph{absolutely continuous} with respect
    to $\mu$, denoted by $\nu\ll\mu$, if
    \[
        |\mu|(A)=0
        \implies |\nu|(A)=0
        \qquad
        \text{for all }A\in\mathcal{B}(X).
    \]

    The measures $\mu$ and $\nu$ are called
    \emph{mutually singular}, denoted by
    $\mu\perp\nu$, if there exists
    $A\in\mathcal{B}(X)$ such that
    \[
        |\mu|(A)=0,
        \qquad
        |\nu|(X\setminus A)=0.
    \]
\end{definition}

The following results establish the
decomposition of a signed measure into
absolutely continuous and singular parts,
as well as the existence of a density for
the absolutely continuous part.

\begin{theorem}[Lebesgue Decomposition and
                  Radon--Nikodym]
    Let $\mu$ be a $\sigma$-finite positive
    measure and $\nu$ a finite signed
    measure on $(X,\mathcal{B}(X))$.

    Then there exist unique finite signed
    measures $\nu_{\mathrm{ac}}$ and
    $\nu_{\mathrm{s}}$ such that
    \[
        \nu=\nu_{\mathrm{ac}}+\nu_{\mathrm{s}},
        \qquad
        \nu_{\mathrm{ac}}\ll\mu,
        \qquad
        \nu_{\mathrm{s}}\perp\mu.
    \]

    Furthermore, there exists a function
    $f\in L^1(X,\mu)$, unique
    $\mu$-almost everywhere, satisfying
    \[
        \nu_{\mathrm{ac}}(A)
        =\int_A f\,\mathrm{d}\mu
        \qquad
        \text{for all }A\in\mathcal{B}(X).
    \]
    The function $f$ is called the
    \emph{Radon--Nikodym derivative} of
    $\nu_{\mathrm{ac}}$ with respect to $\mu$
    and is denoted by
    \[
        f=\frac{\mathrm{d}\nu_{\mathrm{ac}}}
                {\mathrm{d}\mu}.
    \]
\end{theorem}

In particular, if $\nu\ll\mu$, the singular
part vanishes and
$\mathrm{d}\nu=f\,\mathrm{d}\mu$.
When $\mu$ is the Lebesgue measure
$\mathcal{L}^d$, the function $f$ is
called the density of $\nu$.

We conclude by introducing the spaces
of measures used in the following chapters.

\begin{definition}[Spaces of Measures]
    Let $\mathcal{M}(X)$ denote the space
    of finite signed Radon measures on $X$,
    equipped with the total variation norm.
    We further define
    \[
        \mathcal{M}_+(X)
        :=\{\mu\in\mathcal{M}(X):
             \mu\geq 0\}
    \]
    and
    \[
        \mathcal{P}(X)
        :=\{\mu\in\mathcal{M}_+(X):
             \mu(X)=1\}.
    \]
    For $m\in\mathbb{N}$, the space of
    finite $\mathbb{R}^m$-valued Radon
    measures is denoted by
    \[
        \mathcal{M}(X;\mathbb{R}^m)
        :=\bigl(\mathcal{M}(X)\bigr)^m.
    \]
\end{definition}

The space $\mathcal{M}(X)$ is a vector
space, while $\mathcal{M}_+(X)$ is a
convex cone and $\mathcal{P}(X)$ is a
convex subset. Indeed, for
$\mu,\nu\in\mathcal{P}(X)$ and
$\lambda\in[0,1]$,
\[
    \lambda\mu+(1-\lambda)\nu
    \in\mathcal{P}(X).
\]

Vector-valued measures allow us to
represent quantities such as the
transport flux. In particular, an
integrable vector field
$J:X\to\mathbb{R}^m$ induces a
vector-valued measure, provided that
$X\subseteq\mathbb{R}^d$.

\subsection{Foundations of Functional Analysis}

We now introduce the functional analytic concepts required
to study convergence, compactness and weak formulations of
partial differential equations. These notions provide the
basis for the variational formulation of the optimal
transport problem.

%\subsubsection{Normed Spaces and Dual Spaces}

\begin{definition}[Normed, Banach and Hilbert Spaces]
    A vector space $V$ over $\mathbb{R}$ equipped with
    a norm $\|\cdot\|_V$ is called a \emph{normed
    vector space}. If $V$ is complete with respect to
    the metric induced by its norm, it is called a
    \emph{Banach space}.

    A complete inner product space $(H,\langle
    \cdot,\cdot\rangle_H)$ is called a
    \emph{Hilbert space}, with induced norm
    \[
        \|u\|_H=\sqrt{\langle u,u\rangle_H}.
    \]
\end{definition}

\begin{definition}[Dual Space]
    Let $V$ be a normed vector space. Its
    \emph{dual space} is defined by
    \[
        V^*:=\{F:V\to\mathbb{R}
        \mid F\text{ is linear and continuous}\}.
    \]
    The dual norm is given by
    \[
        \|F\|_{V^*}
        :=\sup_{\|v\|_V\leq 1}|F(v)|.
    \]
    For $F\in V^*$ and $v\in V$, we denote
    the duality pairing by
    $\langle F,v\rangle_{V^*\times V}$.
\end{definition}

In particular, the Riesz representation theorem
identifies every Hilbert space $H$ with its
continuous dual $H^*$ through its inner product.
This identification will be used when considering
weak formulations of partial differential equations.

\begin{definition}[Weak and Weak-* Convergence]
    Let $V$ be a Banach space.

    A sequence $(u_n)_{n\in\mathbb{N}}\subset V$
    converges \emph{weakly} to $u\in V$,
    denoted by $u_n\rightharpoonup u$, if
    \[
        F(u_n)\longrightarrow F(u)
        \qquad\text{for every }F\in V^*.
    \]

    A sequence $(F_n)_{n\in\mathbb{N}}\subset V^*$
    converges \emph{weakly-*} to $F\in V^*$,
    denoted by $F_n\overset{*}{\rightharpoonup}F$,
    if
    \[
        \langle F_n,v\rangle_{V^*,V}
        \longrightarrow
        \langle F,v\rangle_{V^*,V}
        \qquad\text{for every }v\in V.
    \]
\end{definition}

Strong convergence in norm implies weak
convergence. The converse does not hold in
general. The following results provide
important compactness properties.

\begin{theorem}[Weak Compactness in Reflexive Spaces]
    Let $V$ be a reflexive Banach space.
    Then every bounded sequence in $V$
    admits a weakly convergent subsequence.
\end{theorem}

\begin{theorem}[Banach--Alaoglu]
    Let $V$ be a normed vector space. Then
    the closed unit ball in $V^*$ is compact
    with respect to the weak-* topology.

    If $V$ is separable, every bounded
    sequence in $V^*$ admits a weakly-*
    convergent subsequence.
\end{theorem}

These results are fundamental in variational
problems, where bounded minimizing sequences
can be used to extract convergent subsequences.

%\subsubsection{Narrow Convergence and Compactness}

We now specialize the concept of convergence
to measures. Throughout this part, let $X$
be a Polish space, i.e., a topological space
whose topology is induced by a complete,
separable metric.

\begin{definition}[Narrow Convergence]
    A sequence $(\mu_n)_{n\in\mathbb{N}}$
    of probability measures in $\mathcal{P}(X)$
    is said to converge \emph{narrowly} to
    $\mu\in\mathcal{P}(X)$ if
    \[
        \int_X\varphi\,\mathrm{d}\mu_n
        \longrightarrow
        \int_X\varphi\,\mathrm{d}\mu
        \qquad
        \forall\,\varphi\in C_b(X),
    \]
    where $C_b(X)$ denotes the space of
    bounded continuous functions on $X$.
\end{definition}

For finite Radon measures on a locally compact
space, convergence against test functions
in $C_0(X)$ is commonly referred
to as vague convergence. By the Riesz
representation theorem, this corresponds to
weak-* convergence in the dual space
$\mathcal{M}(X)=C_0(X)^*$.

\begin{definition}[Tightness]
    A family $\mathcal{K}\subseteq\mathcal{P}(X)$
    is called \emph{tight} if for every
    $\varepsilon>0$ there exists a compact
    set $K_\varepsilon\subseteq X$ such that
    \[
        \mu(X\setminus K_\varepsilon)
        <\varepsilon
        \qquad
        \forall\,\mu\in\mathcal{K}.
    \]
\end{definition}

\begin{theorem}[Prokhorov's Theorem]
    Let $X$ be a Polish space. A family
    $\mathcal{K}\subseteq\mathcal{P}(X)$
    is relatively compact with respect to
    narrow convergence if and only if
    $\mathcal{K}$ is tight.
\end{theorem}

An important consequence of narrow convergence
is the following lower semicontinuity property.

\begin{theorem}[Portmanteau Theorem,
                 Lower Semicontinuity]
    Let $\mu_n,\mu\in\mathcal{P}(X)$ with
    $\mu_n$ converging narrowly to $\mu$.
    Then, for every lower semicontinuous
    function $\varphi:X\to[0,\infty]$,
    \[
        \int_X\varphi\,\mathrm{d}\mu
        \leq
        \liminf_{n\to\infty}
        \int_X\varphi\,\mathrm{d}\mu_n.
    \]
\end{theorem}

Together, compactness and lower semicontinuity
are essential tools for establishing the
existence of minimizers in optimal transport
and related variational problems.

    :=\int_\Omega uv\,\mathrm{d}x.
\]

\begin{definition}[Weak Derivative]
    Let $u\in L^1_{\mathrm{loc}}(\Omega)$.
    A function $v_i\in L^1_{\mathrm{loc}}(\Omega)$
    is called the \emph{weak derivative}
    of $u$ with respect to $x_i$ if
    \[
        \int_\Omega u\,\partial_i\varphi
        \,\mathrm{d}x
        =
        -\int_\Omega v_i\varphi
        \,\mathrm{d}x
    \]
    for all $\varphi\in C_c^\infty(\Omega)$.
    We write $\partial_i u=v_i$.
\end{definition}

Weak derivatives extend classical
differentiation to functions that need not
be continuously differentiable. They are
the basis for the definition of Sobolev spaces.

\begin{definition}[Sobolev Spaces]
    For $1\leq p\leq\infty$, the Sobolev
    space $W^{1,p}(\Omega)$ is defined by
    \[
        W^{1,p}(\Omega)
        :=\{u\in L^p(\Omega):
        \partial_i u\in L^p(\Omega),
        \ i=1,\ldots,d\},
    \]
    where the derivatives are understood
    in the weak sense.

    For $1\leq p<\infty$, it is equipped
    with the norm
    \[
        \|u\|_{W^{1,p}(\Omega)}
        :=
        \left(
        \|u\|_{L^p(\Omega)}^p+
        \|\nabla u\|_{L^p(\Omega)}^p
        \right)^{1/p}.
    \]

    In particular, we write
    \[
        H^1(\Omega):=W^{1,2}(\Omega).
    \]
\end{definition}

The space $H^1(\Omega)$ is a Hilbert space
with inner product
\[
    \langle u,v\rangle_{H^1(\Omega)}
    =
    \int_\Omega
    \left(uv+\nabla u\cdot\nabla v\right)
    \,\mathrm{d}x.
\]

We denote by $H_0^1(\Omega)$ the closure

where the divergence is understood in
the distributional sense.

These spaces allow the differential
constraints arising in optimal transport
to be interpreted in a weak sense, even
when classical derivatives do not exist.

\subsection{Foundations of Convex Analysis and
Variational Methods}

We introduce the main concepts of convex analysis
and variational methods required for the formulation
and solution of constrained optimization problems.
In particular, we discuss the existence of minimizers,
convex duality and the characterization of solutions
through saddle points and Lagrange multipliers.

%\subsubsection{The Direct Method}

The direct method of the calculus of variations
provides a fundamental approach to establishing
the existence of minimizers.

\begin{definition}[Coercivity and Lower
Semicontinuity]
    Let $V$ be a Banach space and
    $F:V\to(-\infty,\infty]$ a functional.

    The functional $F$ is called
    \emph{coercive} if
    \[
        \|u\|_V\to\infty
        \quad\Longrightarrow\quad
        F(u)\to\infty.
    \]

    It is called \emph{sequentially weakly
    lower semicontinuous} if
    \[
        u_n\rightharpoonup u
        \quad\Longrightarrow\quad
        F(u)\leq\liminf_{n\to\infty}F(u_n).
    \]
\end{definition}
%TODO proper?
\begin{theorem}[Direct Method]
    Let $V$ be a reflexive Banach space and
    $F:V\to(-\infty,\infty]$ be proper,
    bounded from below, coercive and
    sequentially weakly lower semicontinuous.

    Then there exists $u^*\in V$ such that
    \[
        F(u^*)=\inf_{u\in V}F(u).
    \]
\end{theorem}

The underlying idea is to construct a minimizing
sequence $(u_n)_{n\in\mathbb{N}}$. Coercivity
implies its boundedness, while reflexivity
allows the extraction of a weakly convergent
subsequence. Weak lower semicontinuity then
ensures that its limit is a minimizer.

For constrained problems, the admissible set
must additionally be compatible with the
chosen convergence, for instance by being
weakly sequentially closed.

%\subsubsection{Convex Functions and Conjugation}

\begin{definition}[Convex Functional]
    Let $V$ be a real vector space.
    A functional $F:V\to(-\infty,\infty]$
    is called \emph{convex} if
    \[
        F(\lambda u+(1-\lambda)v)
        \leq \lambda F(u)+(1-\lambda)F(v)
    \]
    for all $u,v\in V$ and
    $\lambda\in[0,1]$.

    It is called \emph{strictly convex} if
    the inequality is strict whenever
    $u\neq v$, $\lambda\in(0,1)$ and
    $F(u),F(v)<\infty$.
\end{definition}

Convexity is particularly useful in
optimization, since every local minimizer
of a convex functional is also a global
minimizer. Moreover, a strictly convex
functional admits at most one minimizer
on a convex admissible set.

\begin{definition}[Subdifferential]
    Let $V$ be a Banach space and
    $F:V\to(-\infty,\infty]$ be proper
    and convex. The \emph{subdifferential}
    of $F$ at $u\in V$ is
    \[
        \partial F(u):=
        \left\{
            p\in V^*:
            F(v)\geq F(u)
            +\langle p,v-u\rangle_{V^*\times V}
            \quad\forall v\in V
        \right\},
    \]
    provided that $F(u)<\infty$.
    Otherwise, $\partial F(u)=\emptyset$.
\end{definition}

For a differentiable convex functional,
the subdifferential reduces to its derivative.
In particular, a point $u^*$ minimizes
a proper convex functional $F$ if and
only if
\[
    0\in\partial F(u^*).
\]

\begin{definition}[Convex Conjugate]
    Let $V$ be a Banach space and
    $F:V\to(-\infty,\infty]$ a proper
    functional. Its \emph{convex conjugate}
    $F^*:V^*\to(-\infty,\infty]$ is defined by
    \[
        F^*(p):=
        \sup_{u\in V}
        \left\{
            \langle p,u\rangle_{V^*\times V}
            -F(u)
        \right\}.
    \]
\end{definition}

The convex conjugate is always convex and
lower semicontinuous in the weak-* topology.
Its relation to the original functional is
characterized by the following results.

\begin{theorem}[Fenchel--Young Inequality]
    Let $F:V\to(-\infty,\infty]$ be proper
    and convex. Then
    \[
        F(u)+F^*(p)
        \geq \langle p,u\rangle_{V^*\times V}
    \]
    for all $u\in V$ and $p\in V^*$.

    Equality holds if and only if
    $p\in\partial F(u)$.
\end{theorem}

\begin{theorem}[Fenchel--Moreau]
    Let $F:V\to(-\infty,\infty]$ be a
    proper, convex and lower semicontinuous
    functional on a Banach space $V$.
    Then
    \[
        F^{**}=F,
    \]
    where the canonical embedding of
    $V$ into its bidual is understood.
\end{theorem}

We consider a general convex optimization
problem involving a linear operator.

Let $V$ and $Y$ be Banach spaces,
$A:V\to Y$ a bounded linear operator,
and $F:V\to(-\infty,\infty]$ and
$G:Y\to(-\infty,\infty]$ proper, convex,
lower semicontinuous functionals.
The primal problem is
\[
    \inf_{u\in V}\left\{F(u)+G(Au)\right\}.
\]

\begin{theorem}[Fenchel--Rockafellar Duality]
    Suppose there exists
    $u_0\in\operatorname{dom}(F)$ such
    that $G$ is continuous at $Au_0$.

    Then
    \[
        \inf_{u\in V}
        \left\{F(u)+G(Au)\right\}
        =
        \max_{p\in Y^*}
        \left\{
            -F^*(-A^*p)-G^*(p)
        \right\},
    \]
    provided that the primal infimum is
    finite, where $A^*:Y^*\to V^*$
    denotes the adjoint operator.
\end{theorem}

The left-hand side is called the
\emph{primal problem}, while the
right-hand side is its \emph{dual problem}.
The equality of their optimal values is
called \emph{strong duality}.

Without additional regularity assumptions,
only \emph{weak duality} is guaranteed,
meaning that the dual optimal value
does not exceed the primal optimal value.

Convex duality plays an important role in
optimal transport, where the nonlinear
transport cost can be expressed through
its convex conjugate.

%\subsubsection{Saddle Points and
%Lagrange Multipliers}

Consider the constrained optimization problem
\[
    \min_{u\in V}F(u)
    \qquad\text{subject to}\qquad Au=b,
\]
where $A:V\to Y$ is bounded and linear,
$b\in Y$, and $F$ is proper and convex.

\begin{definition}[Lagrangian]
    The \emph{Lagrangian} associated with
    the constrained optimization problem is
    \[
        \mathcal{L}(u,p)
        :=
        F(u)+\langle p,Au-b\rangle_{Y^*,Y},
    \]
    where $p\in Y^*$ is called a
    \emph{Lagrange multiplier}.
\end{definition}

\begin{definition}[Saddle Point]
    A pair $(u^*,p^*)\in V\times Y^*$
    is called a \emph{saddle point} of
    $\mathcal{L}$ if
    \[
        \mathcal{L}(u^*,p)
        \leq \mathcal{L}(u^*,p^*)
        \leq \mathcal{L}(u,p^*)
    \]
    for all $u\in V$ and $p\in Y^*$.
\end{definition}

A saddle point characterizes a primal
optimal solution together with an
associated dual multiplier. Its existence
is closely related to strong duality and
dual attainment.

\begin{theorem}[Saddle Point Optimality
Conditions]
    Let $F:V\to(-\infty,\infty]$ be
    proper and convex, and let
    $\mathcal{L}$ be defined as above.

    A pair $(u^*,p^*)$ is a saddle point
    of $\mathcal{L}$ if and only if
    \[
        \begin{cases}
            Au^*=b,\\[2mm]
            0\in\partial F(u^*)+A^*p^*.
        \end{cases}
    \]
\end{theorem}

These relations are the Karush--Kuhn--Tucker
(KKT) conditions for the equality-constrained
convex problem. The first condition expresses
primal feasibility, while the second describes
stationarity with respect to the primal
variable.

For numerical optimization, the Lagrangian
can be augmented by a quadratic penalty term.

\begin{definition}[Augmented Lagrangian]
    For a Hilbert space $Y$ and a parameter
    $r>0$, the \emph{augmented Lagrangian}
    is defined by
    \[
        \mathcal{L}_r(u,p)
        :=
        F(u)
        +\langle p,Au-b\rangle_Y
        +\frac{r}{2}\|Au-b\|_Y^2.
    \]
\end{definition}

The additional quadratic term penalizes
violations of the constraint without
changing the objective on the feasible set.
The augmented Lagrangian provides the
theoretical foundation for multiplier
methods and primal-dual algorithms,
including the ALG2 approach used later
in this thesis.
