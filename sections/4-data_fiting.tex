For our modeling purposes we modify the constraints and rewrite the initial Benamou-Brenier formulation as 

	$$\inf_{(v_t,\rho_t)} \int_0^1\int_\Omega \frac{|v_t|^2}{2}\mathrm{d}\rho_t(x)\mathrm{d}t$$
	$$\text{s.t.}\begin{cases}\partial_t\rho_t + \nabla \cdot \left(\rho_t v_t + \rho_tv_t^{wind}\right) = \alpha_{dif}\Delta\rho_t + f_t\quad \text{on }\Omega\times[0,1],\\
	\left(\rho_tv_t + \rho_tv_t^{wind} -\alpha_{dif}\nabla\rho_t\right) \cdot n = 0\quad \text{on } \partial\Omega\times[0,1],\\

$$
-\alpha_{dif}\Delta\rho_t = \nabla \cdot (-\alpha_{dif}\nabla\rho_t ),\quad \nabla\rho_t = \rho_t \nabla \log\rho_t

\textbf{Step 2}: \underline{Update $(\rho_t^*, J_t^*, f_t^*, \nabla\rho_t^*)$}

The new iterates are defined as the solution of 
\begin{align*}
	\inf_{(\rho_{m+1}^*, J_{m+1}^*, f_{m+1}^*, g_{m+1}^*)} &\int_{((0,1)\times\Omega)\setminus\Gamma} (F^{\alpha_f,\alpha_{dif}})^*(\rho_t^*, J_t^*, f_t^*,g_t^*)\,\mathrm{d}x\mathrm{d}t + \int_\Gamma (\overline{F}^{\alpha_f,\alpha_{dif}})^*(\rho_t^*,J_t^*,f_t^*,g_t^*)\,\mathrm{d}x\mathrm{d}t\\
															&- \langle\rho_t^*,\rho_m + r_1(\partial_t\phi_{m+1} + \nabla\phi_{m+1} \cdot v_t^{wind} - \nabla \cdot \vartheta_{m+1} \rangle+ \frac{r_1}{2} \lVert \rho_t^*\rVert^2\\
	&-\langle J_t^*, J_m + r_1\nabla\phi_{m+1}\rangle+ \frac{r_1}{2} \lvert J_t^*\rvert^2-\langle f_t^*, f_m + r_1\phi_{m+1}\rangle+ \frac{r_1}{2} \lvert f_t^*\rvert^2\\
	&-\langle g_t^*, g_m - r_1\vartheta_{m+1}\rangle+ \frac{r_1}{2}\lvert g_t^*\rvert^2
\end{align*}
where terms constant in $(\rho_{m+1}^*,J_{m+1}^*, f_{m+1}^*, g_{m+1}^*)$ are neglected. We optimize this problem in $((0,1)\times \Omega)\setminus\Gamma$ and in $\Gamma$ independently.
Taking the variables defined above, the minimization in the former reads as
\begin{equation}\label{eq:Update2}
\begin{aligned}
	\inf_{(\rho_{m+1}^*, J_{m+1}^*, f_{m+1}^*, g_{m+1}^*)} &\iota_{\left\{ \rho_t^* + \frac{1}{2} \left(|J_t^*| + \frac{|f_t^*|}{\alpha_f} + \frac{|g_t|}{\alpha_{dif}}^2\right)\leq 0 \right\}} - r_1\langle \eta_\Omega , (\rho_{m+1}^*, J_{m+1}^*, f_{m+1}^*, g_{m+1}^*)\rangle \\
																	 +&\frac{r_1}{2} (|\rho_{m+1}^*|^2 + |J_{m+1}^*|^2 + |f_{m+1}^*|^2 + |g_{m+1}^*|^2)
\end{aligned}
\end{equation}
with the optimal solution $\eta_\Omega$, if it is admissible. If not, we introduce another Lagrange multiplier $\mu >0$ and since the problem has a constant term with respect to $\eta_\Omega$ we can rewrite as
$$
\inf_{(\rho_{m+1}^*, J_{m+1}^*, f_{m+1}^*, g_{m+1}^*)} \frac{1}{2}|z_{m+1}^* - \eta_\Omega|_\Omega^2
$$
The new Lagrangian for this problem is
\begin{align*}
	\mathcal{L}(\rho_t^*, J_t^*, f_t^*, g_t^*, \mu) &= \frac{1}{2}(|\rho^*-\eta_{\rho^*}|^2 + |J^* - \eta_{J^*}|^2 + |f^*-\eta_{f^*}|^2 + |g^* - \eta_{g^*}|^2) \\&+ \mu\rho^* + \frac{\mu}{2}\left(|J^*|^2 + \frac{|f^*|^2}{\alpha_f} + \frac{|g^*|}{\alpha_{dif}}\right)
\end{align*}
%and stationarity conditions lead to 
%$$
%\begin{cases}
%\rho^* = \eta_{\rho^*} -\mu\\
%J^* = \frac{\eta_{J^*}}{1+\mu}\\
%f^* = \frac{\eta_{f^*}}{1+\frac{\mu}{\alpha_f}}\\
%\nabla\rho^* = \frac{\eta_{\nabla\rho^*}}{1+\frac{\mu}{\alpha_{dif}}}
%\end{cases}
%$$
Define
$$
x:= |J^*|,\quad y:= |f^*|,\quad u:=|g^*|
$$
and rewrite the active constraint as 
$$
\rho^* = -\frac{1}{2}\left(x^2+ \frac{y^2}{\alpha_f}+\frac{u^2}{\alpha_{dif}}\right).
$$
Substituting this expression into 
$$
\min_{x,y,u} \frac{1}{2}\left[ \left( -\frac{1}{2}\left(x^2 + \frac{y^2}{\alpha_f}+\frac{u^2}{\alpha_{dif}}\right) - \eta_{\rho^*}\right)^2 + (x-\lvert \eta_{J^*}\rvert)^2 + (y-\lvert\eta_{f^*}\rvert)^2 + (u- \lvert \eta_{g^*}\rvert)^2 \right]
$$
and taking the derivatives with respect to $x,y$ and $u$ yields the equations
\begin{align*}
	0&=x^3 + \frac{1}{\alpha_f}xy^2 + \frac{1}{\alpha_{dif}}xu^2 + 2(1+\eta_{\rho^*})x - 2\lvert\eta_{J^*}\rvert \\
	0&=y^3 + \alpha_fx^2y + \frac{\alpha_f}{\alpha_{dif}}yu^2 + 2(\alpha_f^2 + \alpha_f\eta_{\rho^*})y - 2\alpha_f^2|\eta_{f^*}|\\
	0&= u^3 + \alpha_{dif}x^2u + \frac{\alpha_{dif}}{\alpha_f}y^2u + 2\alpha_{dif}(\alpha_{dif}+\eta_{\rho^*})u- 2\alpha_{dif}^2 |\eta_{g^*}|.
\end{align*}
Solving these equations for $x,y,u$ we recover
$$
\begin{cases}
	\rho^* = -\frac{1}{2}\left(x^2 + \frac{y^2}{\alpha_f}+\frac{u^2}{\alpha_{dif}}\right),\\
	J^* = x\frac{\eta_{J^*}}{\lvert \eta_{J^*}\rvert},\\
	f^* = y\frac{\eta_{f^*}}{|\eta_{f^*}|},\\
	g^* = u\frac{\eta_{g^*}}{\lvert \eta_{g^*}\rvert},
\end{cases}
$$
whenever the denominators are non-zero for an analytical solution. This can numerically be solved also by using Newton's method. On $\Gamma$ we can solve the system that is obtained by calculating the gradient of \eqref{eq:Update2}
$$
\begin{pmatrix}
	\rho_{data}+ 2\left(\frac{1}{\alpha_{inv}}-\frac{\alpha_{inv}}{2}\right)\left(\rho^* + \frac{1}{2}|J^*|^2 + \frac{1}{\alpha_f}|f^*|^2 + \frac{1}{2\alpha_{dif}}|g^*|^2\right) + r_1(\rho^* - \eta_{\rho^*})\\
	\rho_{data}J^* + 2\left(\frac{1}{\alpha_{inv}}-\frac{\alpha_{inv}}{2}\right) \left(\rho^* + \frac{1}{2}|J^*|^2 + \frac{1}{\alpha_f}|f^*|^2 + \frac{1}{2\alpha_{dif}}|g^*|\right)J^* + r_1(J^* - \eta_{J^*})\\
	\frac{1}{\alpha_f}\rho_{data}f^* + \frac{1}{\alpha_f}\left(\frac{1}{\alpha_{inv}}-\frac{\alpha_{inv}}{2}\right) \left(\rho^* + \frac{1}{2}|J^*|^2 + \frac{1}{\alpha_f}|f^*|^2 + \frac{1}{2\alpha_{dif}}|g^*|\right)f^*  + r_1(f^* - \eta_{f^*})\\
	\frac{1}{\alpha_{dif}}\rho_{data}g^* + \frac{1}{\alpha_{dif}}\left(\frac{1}{\alpha_{inv}}-\frac{\alpha_{inv}}{2}\right) \left(\rho^* + \frac{1}{2}|J^*|^2 + \frac{1}{\alpha_f}|f^*|^2 + \frac{1}{2\alpha_{dif}}|g^*|^2\right) g^* + r_1(g^* - \eta_{g^*})
\end{pmatrix} = \begin{pmatrix}
0\\0\\0\\0
\end{pmatrix}
$$


\textbf{Step 3:} \underline{Update $(\rho_t, J_t, f_t, g_t)$}

As in step 2, we neglegt terms constant to the respective variables and need to solve 
\begin{align*}
	\sup_{(\rho_t,J_t, f_t, g_t)}&\langle(\partial_t\phi_t + \nabla\phi_t \cdot v_t^{wind} - \rho_t^* - \nabla \cdot \vartheta_t, \nabla\phi_t - J_t^*, \phi_t - f_t^*, - \vartheta_t -g_t^*), (\rho_t, J_t, f_t, g_t)\rangle%\\
	%TODO Das noch abklären &- G(\phi_t) + \vartheta_t \left(\int_\Omega d\rho_1 - \int_\Omega d\rho_0 - \int_0^1\int_\Omega f_t\,\mathrm{d}x\mathrm{d}t\right)
\end{align*}
which is done by a single gradient descent step in each variable with step size $r_1$, i.e. 
\begin{align*}
	&\rho_{m+1} := \rho_m + r_1 (\partial_t\phi_{m+1} + \nabla\phi_{m+1} \cdot v_t^{wind} - \rho_{m+1}^* - \nabla \cdot \vartheta_{m+1})\\
	&J_{m+1} := J_m + r_1(\nabla\phi_{m+1} - J_{m+1}^*)\\
	&f_{m+1} := f_m + r_1(\phi_{m+1} - f_{m+1}^*)\\% + \vartheta_t\\
	&g_{m+1} := g_m + r_1 (-\vartheta_{m+1} - g_{m+1}^*).
\end{align*}

\begin{algorithm}
	\caption{Augmented Lagrangian Method with diffusion term}
	\begin{algorithmic}
		\While{$m< it_max^{ALG}$ and $err^{ALG} > tol$}

		Solve $A(\phi_{m+1}) = F^m$ for $\phi_{m+1}$

		Update $\vartheta_{m+1}$

		$\eta_{\rho^*} := \partial_t\phi_{m+1} + \nabla\phi_{m+1} \cdot v_t^{wind} + \frac{1}{r_1}\rho_m$

		$\eta_{J^*} := \nabla\phi_{m+1} + \frac{1}{r_1}J_m$

		$\eta_{f^*} := \phi_{m+1} + \frac{1}{r_1}f_m$

		$\eta_{\nabla\phi^*} := \frac{1}{r_1}\nabla\rho_m$

		$\eta_\Omega := (\eta_{\rho^*},\eta_{J^*},\eta_{f^*},\eta_{\nabla\rho^*})$

		\ForAll{$(t,x) \in(0,1)\times\Omega\setminus \Gamma$}
		\If{$\eta_{\rho^*}+\frac{1}{2}\left(|\eta_{J^*}|^2 + \frac{|\eta_{f^*}|^2}{\alpha_f} +\frac{|\eta_{\nabla\rho^*}|^2}{\alpha_{dif}}\right)\leq 0$}
			
		$(\rho_{m+1}^*,J_{m+1}^*,f_{m+1}^*,\nabla\rho_{m+1}^*) = (\eta_{\rho^*},\eta_{J^*},\eta_{f^*},\eta_{\nabla\rho^*})$
			 \Else

			 Solve nonlinear system for $x,y,u\geq 0$

			 $\rho_{m+1}^*\gets -\frac12\left(x^2+\frac{y^2}{\alpha_f}+\frac{u^2}{\alpha_{dif}}\right)$

			 $J_{m+1}^* \gets x\frac{\eta_{J^*}}{|\eta_{J^*}|}$

			 $f_{m+1}^* \gets y\frac{\eta_{f^*}}{|\eta_{f^*}|}$

			 $\nabla\rho_{m+1}^*\gets u \frac{\eta_{\nabla\rho^*}}{|\eta_{\nabla\rho^*}|}$
			 \EndIf
	\EndFor
	\ForAll{$(t,x)\in\Gamma$}
	
	Solve $\nabla_z\left[(F^{\alpha_f,\alpha_{dif},\alpha_{inv}}+\frac{r_1}{2})^* |z^*-\eta_{\Omega}|^2\right] = 0$
		for $z^* = (\rho^*,J^*,f^*,\nabla\rho^*)$

		Set $z_{m+1}^*\gets z^*$
		\EndFor

		Update $(\rho_{m+1},J_{m+1},f_{m+1},\nabla\rho_{m+1})$ by

		$\rho_{m+1}\gets\rho_m+r_1\left(\partial_t\phi_{m+1}+v_t^{wind}\cdot\nabla\phi_{m+1}-\rho_{m+1}^*\right)$

		$J_{m+1}\gets J_m + r_1\left(\nabla\phi_{m+1}-J_{m+1}^*\right)$

		$f_{m+1}\gets f_m+r_1\left(\phi_{m+1}-f_{m+1}^*\right) + \vartheta_{m+1}$

		$\nabla\rho_{m+1}\gets\nabla\rho_m - r_1\nabla\rho_{m+1}^*$


		Compute while-loop conditions

		$err^{ALG}\gets \max\left\{\left|\partial_t\phi_{m+1}+v_t^{wind}\cdot\nabla\phi_{m+1}-\rho_{m+1}^*\right|_\infty, \left|\nabla\phi_{m+1} - J_{m+1}^*\right|_\infty, \left|\phi_{m+1} - f_{m+1}^*\right|_\infty, \left|\nabla\rho_{m+1}^*\right|_\infty\right\}$

		$m\gets m+1$
	\EndWhile
	\end{algorithmic}
\end{algorithm}
