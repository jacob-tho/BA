\section{Data Fitting}

For our modeling purposes we modify the constraints and rewrite the initial Benamou-Brenier formulation as

	$$\inf_{(v_t,\rho_t)} \int_0^1\int_\Omega \frac{|v_t|^2}{2\rho}\mathrm{d}x\mathrm{d}t$$
	$$s.t.\begin{cases}\partial_t\rho_t + \nabla \cdot (\rho_t v_t + \rho_tv_t^{wind}) = \alpha_{dif}\Delta\rho_t + f_t\quad \text{on }\Omega\times[0,1]\\
	(\rho_tv_t + \rho_tv_t^{wind} -\alpha_{dif}\nabla\rho_t) \cdot n = 0\quad \text{on } \partial\Omega\times[0,1]\\
\rho(0)  = \rho_0,\quad\rho(1) = \rho_1\quad \end{cases}$$

If we recall that 
$$
-\alpha_{dif}\Delta\rho_t = \nabla \cdot (-\alpha_{dif}\nabla\rho_t ),\quad \nabla\rho_t = \rho_t \nabla \log\rho_t
$$
we can reformulate the first constraint as 
\begin{align*}
	&\partial_t\rho_t + \nabla \cdot (\rho_tv_t + \rho_tv_t^{wind}) - \alpha_{dif}\Delta\rho_t = f_t\\
	&\iff \partial_t\rho_t + \nabla \cdot (\underbrace{\rho_t ( v_t - \alpha_{dif}\nabla\log\rho_t)}_{=:J_t} + \rho_tv_t^{wind}) = f_t
\end{align*}
Using this notation and rewriting $v_t$ gives us
$$
\int_0^1\int_\Omega \frac{|J_t|^2}{2\rho_t} \mathrm{d}xdt + \alpha_{dif}\int_0^1\int_\Omega J_t \cdot \nabla\log\rho_t + \frac{\alpha_{dif}^2}{2}\int_0^1\int_\Omega \underbrace{|\nabla\log\rho_t|^2\rho_t}_{=\frac{|\nabla\rho_t|^2}{\rho_t}}dxdt
$$

and we simplyfy (dürfen wir das einfach so???) the whole problem as
$$
\inf_{(\rho_t,J_t)}\int_0^1\int_\Omega \frac{|J_t|^2}{2\rho_t}dxdt + \alpha_{dif}^2\int_0^1\int_\Omega \frac{|\nabla\rho_t|^2}{\rho_t}\rho_tdxdt
$$
$$
s.t.\begin{cases}
	\partial_t\rho_t + \nabla \cdot (\rho_tv_t + \rho_tv_t^{wind} = f_t\quad \text{on } \Omega \times (0,1)\\
	(\rho_tv_t + \rho_tv_t^{wind}) \cdot n = 0 \quad \text{on } \partial\Omega \times (0,1)\\
	\rho_t(0) = \rho_0\\
	\rho_t(1) = \rho_1
\end{cases}
$$

Adding two terms to complete the model, one for allowing creation of mass and one for fitting data for a given curve, we finally receive
\begin{align*}
&\inf_{(f_t, J_t, \rho_t; \rho_0, \rho_1)} \int_0^1\int_\Omega \frac{|J_t|^2}{2\rho_t}dxdt + \alpha_f\int_0^1\int_\Omega \frac{|f_t|^2}{2\rho_t}dxdt \\&+ \alpha_{dif}^2\int_0^1\int_\Omega\frac{|\nabla\rho_t|^2}{\rho_t}dxdt + \frac{\alpha_{in}}{2}\int_0^1\int_{\Gamma}|\rho_t - \rho_{data}|^2dxdt
\end{align*}
$$
s.t. \begin{cases}
	\partial_t\rho_t + \nabla \cdot (J_t + \rho_tv_t^{wind}) = f_t\quad \text{on } \Omega\times(0,1)\\
(J_t + \rho_tv_t^{wind}) \cdot n = 0 \quad \text{on }\partial \Omega \times (0,1)
\end{cases}
$$
and we may impose the condition on the initial and final distribution as above. Using the boundary condition we get for the first constraint
\begin{align*}
&\int_0^1\int_\Omega \partial_t\rho_t + \int_0^1\int_\Omega \nabla \cdot (J_t + \rho_tv_t^{wind} )dxdt - \int_0^1\int_\Omega f_tdxdt\\
&= \int_0^1\frac{d}{dt}\left(\int_\Omega \rho_tdx\right)dt - \int_0^1\int_\Omega f_t dxdt\\
&= \int_\Omega\rho_1 dx - \int_\Omega \rho_0dx - \int_0^1\int_\Omega f_t dxdt = 0
\end{align*}
Here, our goal is also to solve the saddle-point problem, for which we need to calculate the Lagrangian in the first place. Taking a smooth test function $\phi_t: [0,1]\times\Omega$ we obtain
\begin{align*}
	\mathcal{L}(\rho_t,J_t, f_t, \phi_t, \lambda_t) &= \int_0^1\int_\Omega\frac{|J_t|^2}{2\rho_t}dxdt + \alpha_f\int_0^1\int_\Omega \frac{|f_t|^2}{2\rho_t}dxdt \\
													&+\alpha_{dif}^2\int_0^1\int_\Omega\frac{|\nabla\rho_t|^2}{\rho_t}dxdt\\
													&+\frac{\alpha_{inv}}{2} \int_0^1\int_\Omega|\rho_t-\rho_{data}|^2dxdt\\
													&-\int_0^1\int_\Omega \rho_t\partial_t\phi_t + J_t \cdot \nabla\phi_t + \rho_tv_t^{wind} \cdot \nabla\phi_t + f_t\phi_t dxdt\\
													&+\int_\Omega [\phi_1\rho_1 - \phi_0\rho_0]dx\\
													&+\lambda_t\left(\int_\Omega\rho_1dx - \int_\Omega\rho_0dx - \int_0^1\int_\Omega f_tdxdt\right)
\end{align*}
 and the saddle point problem then is
 $$
 \inf_{(\rho_t, J_t, f_t)}\sup_{(\phi_t, \lambda_t)} \mathcal{L}(\rho_t, J_t, f_t, \phi_t, \lambda_t).
