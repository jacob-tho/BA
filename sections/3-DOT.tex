\section{Optimal Transport Theory}
\label{sec:ot}

In this chapter we lay out the framework in which our mathematical model is presented. We introduce the concept of optimal transport, first the static formulation of Monge-Kantorovich and in combination with the continuity equation the dynamical formulation that is then presented through the lense of the Benamou-Brenier formulation. The goal then is to establish a result on the equivalence of solutions and also introduce a version of the ALG2 algorithm presented in (BB). This helps us understand the approach used to solve the modified variational problem in the next chapter.
\subsection{Static Optimal Transport}
The central question of the problem was proposed by Gaspard Monge in 1781. Given a pile of dirt and an equal sized hole, how can it be filled "optimally" (Tikz Grafik einfügen). This raises questions such as what optimal means in that context and also if the problem can even be solved. 
\begin{definition}[Monge-Problem]
	Let $\mu\in \mathscr{P}(X)$ and $\nu\in\mathscr{P}(Y)$ be two probability measures and $c:X\times Y\to[0,\infty]$ a continuous or lower semi-continuous \textit{cost function}. The \textbf{Monge problem} is to find a measurable map $T:X\to Y$ that minimizes the total transpotation cost
	\begin{equation*}\tag{MP}\label{eq:MP}
		\inf\left\{\int_X c(x,T(x))\, \mathrm{d}\mu(x)\ \middle|\ T_\#\mu = \nu\right\}.
	\end{equation*}
\end{definition}
This initial definition allows for an intuitive interpration of the problem, however it inherently comes with the problem that it may not exist a solution.
\begin{example}
	Consider the case $X=Y=\mathbb{R}^n$ and $c(x,y) = \lVert x-y\rVert^2$, where
	$$
	\mu = \delta_0,\quad \nu = \frac{1}{2}\delta_{-1} + \frac{1}{2}\delta_1.
	$$
	A map $T$ solving \eqref{eq:MP} may fail to exist, since mass concentrated at a single point cannot be split and sent to multiple locations to match $\nu$.
\end{example}

Another issue may be that even if a transport map exists, it may not be unique, if we do not allow for strict regulaztion. These potential problems motivate a \textit{relaxation}, that is, rather than seeking a \textit{transport map} we allow \textit{tansport plans}, which are measures on the product space $X\times Y$ that have correct marginals. This idea was introduced by Leonid Kantorovich about 150 years later and is foundational to the modern theory of optimal transport. First we define the concept of a coupling, which formalizes the restriction on the marginals.
\begin{definition}[Coupling]
	Let $\mu\in\mathscr{P}(X)$ and $\mu\in\mathscr{P}(Y)$ be probability measures. A measure $\gamma\in\mathscr{P}(X\times Y)$ is called a \textbf{coupling} of $\mu$ and $\nu$ if its marginals are $\mu$ and $\nu$, respectively. That is,
	$$
	(\pi_X)_\#\gamma = \mu\quad\text{ and }\quad (\pi_Y)_\#\gamma = \nu,
	$$
	where $\pi_X(x,y) = x$ and $\pi_Y(x,y) = y$ denote the canonical projections onto $X$ and $Y$. The set of all such couplings is denoted by $\Gamma(\mu,\nu)$. Couplings are commonly referred to as \textbf{transport plans}.
\end{definition}

A result that is crucial for upcoming result is the fact, that under suitable conditions, the set $\Gamma$ is non-empty.

\begin{lemma}
	Let $X$ and $Y$ be a Polish spaces and let $\mu\in \mathscr{P}(X)$ and $\nu \in \mathscr{P}(Y)$. Then $\Gamma(\mu,\nu)$ is non-empty, convex and compact with respect to narrow convergence. %$T:X\to Y$ a measurable map such that $T_\#\mu = \nu$. Then $\Gamma(\mu,\nu)$ is non-empty, convex and tight. Moreover, $\Gamma(\mu,\nu)\subset\mathscr{P}(X\times Y)$ is compace under the topology of narrow convergence.
\end{lemma}
\begin{proof}
	%Prokhorok, marginals of tight families are tight.
	Since $\mu\otimes\nu\in\Gamma(\mu,\nu)$, the set of couplings is non-empty. Convexity follow directly from the linearity of the marginal constrains.

	Let $\varepsilon > 0$. Since $\mu$ and $\nu$ are tight, there exist compact sets $K_X\subset X$ and $K_Y\subset Y$ with
	\begin{align*}
		\mu(X\setminus K_X)<\frac{\varepsilon}{2},\quad \nu(Y\setminus K_Y)<\frac{\varepsilon}{2}.
	\end{align*}
	For every $\gamma\in\Gamma(\mu,\nu)$, we have
	\begin{align*}
        &\gamma((X\times Y)
            \setminus(K_X\times K_Y))\\
        &\quad\leq
        \gamma((X\setminus K_X)\times Y)
        +\gamma(X\times(Y\setminus K_Y))\\
        &\quad=
        \mu(X\setminus K_X)
        +\nu(Y\setminus K_Y)
        <\varepsilon.
    \end{align*}
    Hence $\Gamma(\mu,\nu)$ is tight. By Prokhorov's theorem, it is relatively
    compact. Moreover, the marginal maps
    are continuous under narrow convergence,
    so $\Gamma(\mu,\nu)$ is closed. Therefore $\Gamma(\mu,\nu)$ is compact.

\end{proof}

\begin{definition}[Kantorovich Problem]
	Given a lower semi-continuous cost function $c:X\times Y\to [0,\infty]$ and probability measures $\mu\in\mathscr{P}(X),\,\nu\in\mathscr{P}(Y)$, the \textbf{Kantorovich problem} seeks to minimize the total cost over all couplings
	$$
	\inf\left\{\int_{X\times Y} c(x,y)\,\mathrm{d}\gamma(x,y)\ \middle|\ \gamma\in\Gamma(\mu,\nu)\right\}.
	$$
\end{definition}

\begin{remark}
	Since for any feasible transport map $T$ we have 
$$
\int_X c(x,T(x))\, \mathrm{d}\mu(x) = \int_X c\circ (\mathrm{Id}\times T)(x)\,\mathrm{d}\mu(x) = \int_{X\times Y} c(x,y) \, \mathrm{d}\gamma_T(x,y),
$$
which implies that the Monge problem is a special case of the Kantorovich problem. As a consequence it holds
$$
C_M(\mu,\nu)\geq C_K(\mu,\nu),
$$
where $C_M$ and $C_K$ denote the optimal costs of the Monge and Kantorovich problems, respectively.
\end{remark}
For the Dirac example, the unique coupling is $\gamma=\frac{1}{2}\delta_{(0,-1)}+\frac{1}{2}\delta_{(0,1)}$; it is admissible and has quadratic cost $1$.

Under mild assumptions we can proof the existence of a solution for the static optimal transport problem.
\begin{theorem}[Existence of Optimal Transport Plans]
	Let $c:X\times Y\to [0,\infty]$ be lower semi-continuous and let $\mu\in\mathscr{P}(X),\,\nu\in\mathscr{P}(Y)$ be probability measures on Polish spaces. Then there exists a coupling $\overline{\gamma}\in\Gamma(\mu,\nu)$ that solve the Kantorovich problem, i.e.
	$$
	\int_{X\times Y} c(x,y)\,\mathrm{d}\overline{\gamma}(x,y) = \inf_{\gamma\in\Gamma(\mu,\nu)}\int_{X\times Y} c(x,y)\,\mathrm{d}\gamma(x,y).
	$$
\end{theorem}
\begin{proof}
We use the direct method, as explained in ???. Assume that 
$$
\alpha = \inf_{\gamma\in\Gamma(\mu,\nu)}\int_{X\times Y} c(x,y)\,\mathrm{d}\gamma(x,y)< +\infty,
$$
otherwise, every $\gamma\in\Gamma(\mu,\nu)$ is trivially a minimizer. Let $(\gamma_k)_{k\in \mathbb{N}}\subset\Gamma(\mu,\nu)$ be a sequence such that
$$
\int_{X\times Y} c(x,y)\,\mathrm{d}\gamma_k(x,y) \overset{k\to\infty}{\longrightarrow}\alpha.
$$
Since $\Gamma(\mu,\nu)$ is a tight set, the sequence $(\gamma_k)_{k\in \mathbb{N}} \subset \Gamma(\mu,\nu)$ is tight. By Prokhorovs theorem, there exists a subsequence $(\gamma_{k_j})_{j\in \mathbb{N}}$ that converges narrowly to some measure $\overline{\gamma}$ (Hier Resultat staten).

By semi-continuity of the cost functional we have
$$
\alpha = \inf_{\gamma\in\Gamma(\mu,\nu)}\int_{X\times Y} c(x,y)\,\mathrm{d}\gamma(x,y) = \liminf_{j\to\infty} \int_{X\times Y} c(x,y)\,\mathrm{d}\gamma_{k_j}(x,y) \geq \int_{X\times Y} c(x,y)\,\mathrm{d}\overline{\gamma}(x,y).
$$
But since $\overline{\gamma}\in\Gamma(\mu,\nu)$, we also have
$$
\int_{X\times Y}c(x,y)\,\mathrm{d}(x,y)\overline{\gamma}\geq\alpha,
$$
which implies that $\overline{\gamma}$ is a minimizer.

\end{proof}

\subsection{Dynamical Optimal Transport}
\label{sec:dynamical_ot}
The static formulation of optial transport determines how mass is redistributed between two fixed measures, but does not describe its evolution in time. We now introduce particle trajectories, derive the continuity equation and motivate the quadratic density-flux formulation of Benamou and Brenier ~\cite{BB2000}. This section provides a formal derivation for sufficiently regular densities and flows; the rigorous equivalence for measure-valued curves is established in Section~\ref{sec:equivalence_ot}. %The dynamical formulation however looks at a family of densities and a respective flux, linked by a conservation law. We first motivate this formluation for regular flows and subsequently introduce the weak formulation used in the following sections.

Througout this section, let $\Omega\subset \mathbb{R}^n$ be a bounded, convex domain with Lipschitz boundary and let $\rho_0,\rho_1\in L^1(\Omega)$ be probability densities, representing the initial and final densities associated with $\mu$ and $\nu$, respectively. This means that
$$
\mu(A) = \int_A\rho_0(x)\,\mathrm{d}x,\quad\nu(A)=\int_A\rho_1(x)\,\mathrm{d}x
$$
for every Borel set $A\subseteq \Omega$. We describe the evolution of the density by a time-dependent function
$$
\rho_t:[0,1]\times \Omega\to[0,\infty),\quad (t,x)\mapsto\rho_t(x),
$$
with prescribed initial and final densities $\rho_{t=0}=\rho_0$ and $\rho_{t=1} = \rho_1$. To continue the study of the concept, we introduce the central metric.
\begin{definition}[Quadratic Wasserstein Distance]
	Let $\mu,\nu\in\mathscr{P}_2(\mathbb{R}^n)$, where
	$$
	\mathscr{P}_2(\mathbb{R}^n) := \left(\eta\in \mathscr{P}(\mathbb{R}^n) \middle|\int_{\mathbb{R}^n} |x|^2\,\mathrm{d}\eta(x) < \infty\right).
	$$
	The \emph{quadratic Wasserstein distance} is defined by
	\begin{align*}
		W_2(\mu,\nu) := \inf\left\{\left(\int_{\mathbb{R}^n\times \mathbb{R}^n} |x-y|^2\,\mathrm{d}\gamma(x,y)\right)^{\frac{1}{2}}\middle|\gamma\in\Gamma(\mu,\nu)\right\}.
	\end{align*}
\end{definition}
The function $W_2$ defines a metric on $\mathscr{P}_2(\mathbb{R}^n)$ and unlike the total variation norm, it takes into account the spatial distance over which mass is transported. Since $\Omega$ is bounded, the measures $\mu_0$ and $\mu_1$ have finite second moments.

To describe a time-dependent transport, consider a sufficiently regular velocity field $v:[0,1]\times \Omega\to \mathbb{R}^n$. To understand the relationship between the density and the velocity field, we first consider a sufficiently regular flow of particles.

Let $X(t,x)$ denote the position at time $t$ of a particle initially located at $x$. Its trajectory is determined by 
\begin{equation*}
	\begin{cases}\frac{\partial X}{\partial t}(t,x) = v_t(X(t,x)),\\
	X(0,x) = x. \end{cases} \label{eq:particle_flow}
\end{equation*}
Assuming that the flow is well-defined and preserves the domain, conservation of mass implies that for every measurable set $A$ we have
$$
\int_A\rho_0(x)\,\mathrm{d}x = \int_{X(t,A)}\rho_t(y)\,\mathrm{d}y.
$$
In particular, the density at time $t$ is obtained by transporting the initial measure along the flow. In terms of pushforward measures, this reads as 
$$
(X(t,\cdot))_\#\mu = \mu_t,
$$
where $\mu_t$ denotes the measure with density $\rho_t$. At the final time, we require $(X(1,\cdot))_\#\mu = \nu.$

The needed conservation law can be expressed without explicitly tracking individual particles. For a sufficiently smooth test function $\psi$, the pushforward identity gives
$$
\int_\Omega\psi(y)\rho_t(y)\,\mathrm{d}y = \int_\Omega\psi(X(t,x))\rho_0(x)\,\mathrm{d}x.
$$
Differentiating with respect to time and using \eqref{eq:particle_flow}, we obtain
\begin{align*}
	\frac{\mathrm{d}}{\mathrm{d}t}\int_\Omega\psi(y)\rho_t(y)\,\mathrm{d}y
	&= \int\Omega\nabla\psi(X(t,x))\cdot v_t(X(t,x))\rho_0(x)\,\mathrm{d}x\\
	&= \int_\Omega\nabla\psi(y)\cdot v_t(y)\rho_t(y)\,\mathrm{d}y.
\end{align*}
Assuming sufficient regularity and a vanishing normal flux at the boundary, integrating by parts yield
$$
\int_\Omega\left(\partial_t\rho_t + \nabla\cdot(\rho_tv_t)\right)\psi\,\mathrm{d}x = 0.
$$
Since this holds for arbitrary smooth test functions, we obtain the \emph{continuity equation}
\begin{equation*}
	\partial_t\rho_t+\nabla\cdot(\rho_tv_t)=0\label{eq:ce}.
\end{equation*}
This equation expresses conservation of mass, so that changes in the density are caused by the transport of mass through the velocity field.

We now investigate how the Wasserstein distance can be related to the energy of such a flow. For this purpose, consider a sufficiently regular pair $(\rho,v)$ satisfying the continuity equation and the prescribed endpoint conditions. Since 
$$
\gamma := (\mathrm{Id},X(1,\cdot))_\#\mu\in\Gamma(\mu,\nu),
$$
the definition of $W_2$ gives
\begin{align*}
	W_2^2(\mu,\nu)&\leq \int_\Omega|X(1,x)-x|^2\rho_0(x)\,\mathrm{d}x\\
				  &= \int_\Omega\left|\int_0^1\frac{\partial X}{\partial t}(t,x)\,\mathrm{d}t\right|^2\rho_0(x)\,\mathrm{d}x\\
				  &\overset{\text{Jensen}}{\leq} \int_\Omega\int_0^1\left|\frac{\partial X}{\partial t}(t,x)\right|^2\,\mathrm{d}t\,\rho_0(x)\,\mathrm{d}x\\
				  &=\int_0^1\int_\Omega|v_t(X(t,x))|^2\rho_0(x)\,\mathrm{d}x\,\mathrm{d}t.
\end{align*}
Using the pushforward relation once more, we obtain 
\begin{equation*}
	W_2^2(\mu,\nu)\leq \int_0^1\int_\Omega\rho_t(x)|v_t(x)|^2\,\mathrm{d}x\,\mathrm{d}t.\label{eq:wasserstein_bound}
\end{equation*}
Thus, the squared Wasserstein distance is bounded above by the kinetic energy of any sufficiently regular admissible flow. To understand when equality can be achieved, suppose that an optimal transport map $T:\Omega\to\Omega$ exists for the quadratic transport cost. Consider the linear interpolation
\begin{equation*}
	X(t,x):=(1-t)x + tT(x).\label{eq:interpolation}
\end{equation*}
The trajectories' velocity satisfies $\frac{\partial X}{\partial t}(t,x) = T(x) - x$. If the interpolation induces a suitable Eulerian velocity field $v_t$, then for almost every initial position $x$ we have
$$
v_t(X(t,x)) = T(x) -x.
$$
Consequently,
\begin{align*}
	&\int_0^1\int_\Omega\rho_t(y)|v_t(y)|^2\,\mathrm{d}y\,\mathrm{d}t\\
	&\qquad=\int_0^1\int_\Omega |T(x)-x|^2\rho_0(x)\,\mathrm{d}x\,\mathrm{d}t\\
	&\qquad=\int_\Omega |T(x)-x|^2\rho_0(x)\,\mathrm{d}x\\
	&\qquad=W_2^2(\mu,\nu).
\end{align*}
This shows, under the stated assumptions, that the lower bound in \eqref{eq:wasserstein_bound} can be attained by transporting particles along straight lines with constant velocity. These considerations motivate the \emph{dynamical formulation of optimal transport}
\begin{equation}
	\begin{aligned}
		W_2^2(\mu,\nu) &= \inf_{(\rho,v)}
		\int_0^1\int_\Omega\rho_t(x)|v_t(x)|^2\,\mathrm{d}x\,\mathrm{d}t,\\
											 &\text{s.t.}\begin{cases}\partial_t\rho_t + \nabla\cdot(\rho_tv_t)=0,\\
						\rho_{t=0}=\rho_0,\\
						\rho_{t=1}=\rho_1.\end{cases}
	\end{aligned}
	\label{eq:dynamic_ot}
\end{equation}
On a bounded domain, the no-flux condition $(\rho_t v_t)\cdot n = 0$ may be imposed on the spatial boundary. Finally, we introduce the transport flux $J_t := \rho_t v_t.$ This substitution has the advantage, that the continuity equation becomes linear in the variables $(\rho,J)$. Additionally, the kinetic energy takes the form 
$$
\rho_t|v_t|^2 = \frac{|J_t|^2}{\rho_t}
$$
whenever $\rho_t>0$. Using normalization we obtain the \emph{Benamou-Brenier formulation}
\begin{equation}
	\begin{aligned}
		\frac{1}{2}W_2^2(\mu,\nu) &= \inf_{(\rho,J)}
		\int_0^1\int_\Omega\frac{|J_t|^2}{2\rho_t}\,\mathrm{d}x\,\mathrm{d}t,\\
								  &\text{s.t.}\begin{cases}\partial_t\rho_t + \nabla\cdot J_t = 0,\\
						J_t\cdot n = 0\quad\text{on }(0,1)\times\partial\Omega,\\
						\rho_{t=0}=\rho_0,\\
					\rho_{t=1}=\rho_1.\end{cases}
	\end{aligned}
	\label{eq:BB}
\end{equation}
The cost is understood through its lower semicontinuous extension, taking the value zero at $(\rho,J) = (0,0)$ and $+\infty$ whenever $\rho=0$ and $J\neq 0$. The main advantage of this formulation is that the nonlinear term in the constraints disappears, while the objective is convex. 

The preceding argument provides the intuition behind the dynamical formulation under appropiate regularity assumptions. Its rigorous validity for general probability measures, including cases in which no optimal transport map exists, will be established in the following section.

\subsection{Equivalence of the Static and Dynamic Formulation}
In the previous section, the dynamical formulation of optimal transport was motivated by transporting particles along regular trajectories. We now justify this formulation without assuming that a classical flow or an optimal transport map exists. Following the approach of Ambrosio and Savar\'e~\cite{AmbrosioSavare2007}, we first realte absolute continuous curves in the Wasserstein space to the continuity equation and then prove the Benamou-Brenier formula. Since our transport cost is quadratic, we restrict the discussion to $W_2$.

Throughout the proofs, measures are considered on $\mathbb{R}^n$. This allows us to use the standard results for the Wasserstein space $\mathscr{P}_2(\mathbb{R}^n)$. The relation to the bounded, convex domain $\Omega$ introduced in Section ~\ref{sec:dynamical_ot} is discussed at the end of this section.
\begin{definition}[Absolutely Continuous Curve]
	Let $(X,d)$ be a metric space. A curve $\gamma:[0,1]\to X$ is called \emph{absolutely continuous} if there exists $g\in L^1(0,1)$ such that 
	\begin{equation*}
		d(\gamma(s),\gamma(t))\leq\int_s^t g(r)\,\mathrm{d}r\qquad (0\leq s\leq t\leq 1).\label{eq:ac_curve}
	\end{equation*}
	We write $\gamma\in AC([0,1];X)$. If $g$ can be chosen in $L^2(0,1)$, we write $\gamma\in AC^2([0,1];X)$.
\end{definition}
\begin{definition}[Metric Derivative]
	Let $\gamma:[0,1]\to X$ be absolutely continuous. At a time $t\in(0,1)$ for which the following limit exists, the \emph{metric derivative} is defined as 
	\begin{equation*}
		|\gamma'|(t) := \lim_{h\to0}\frac{d(\gamma(t+h),\gamma(t))}{|h|}.\label{eq:metric_derivative}
	\end{equation*}	
\end{definition}

For absolutely continuous curves, the metric derivative exists for almost every $t$ and is the smallest admissible function $g$ in \eqref{eq:ac_curve}. The following estimate will be used repeatedly.

\begin{lemma}[Length Estiamte]
	\label{lem:metric_length}
	If $\gamma\in AC([0,1];X)$, then 
	\begin{equation*}
		d(\gamma(s),\gamma(t))\leq \int_s^t|\gamma'|(r)\,\mathrm{d}r\qquad (0\leq s \leq t \leq 1).\label{eq:metric_bound}
	\end{equation*}
\end{lemma}

\begin{proof}
	The existence and minimality result for the metric derivative already implies the assertion. To see the underlying argument, fix $s\in[0,1]$ and set $f(r):=d(\gamma(s),\gamma(r))$. By the revers triangle inequality,
	$$
	|f(r+h)-f(r)|\leq d(\gamma(r+h),\gamma(r)).
	$$
	The function $f$ is absolutely continuous because $\gamma$ is. At almost every $r$ where both derivatives exist, we have $|f'(r)|\leq |\gamma'|(r)$. Since $f(s)=0$, the fundamental theorem of calculus yields
	$$
	d(\gamma(s),\gamma(t)) = f(t)-f(s)\leq \int_s^t|f'(r)|\,\mathrm{d}r\leq \int_s^t|\gamma'|(r)\,\mathrm{d}r.
	$$
\end{proof}

We next interpret the continuity equation for a curve of measures. If $(\mu_t)_{t\in[0,1]}\subset\mathscr{P}_2(\mathbb{R}^n)$ and $v:(0,1)\times \mathbb{R}^n\to \mathbb{R}^n$ is a Borel velocity field, we say that 
\begin{equation*}
	\partial_t\mu_t+\nabla\cdot(v_t\mu_t) = 0 \label{eq_ce_measure}
\end{equation*}
holds in the distributional sense if 
\begin{equation*}
	\int_0^1\int_{\mathbb{R}^n} \left(\partial_t\varphi(t,x) + v_t(x)\cdot\nabla_x\varphi(t,x)\right)\,\mathrm{d}\mu_t(x)\,\mathrm{d}t = 0\label{eq:ce_weak}
\end{equation*}
for all $\varphi\in C_c^\infty((0,1)\times \mathbb{R}^n)$. We assume $\int_0^1\|v_t\|_{L^2(\mu_t)}^2\,\mathrm{d}t<\infty$. The associated vector-valued flux measure is $J_t:=v_t\mu_t$; if $\mu_t$ has a density $\rho_t$, then $J_t$ has the density $\rho_t v_t$.

\begin{lemma}[Lipschitz Reparametrization]
	\label{lem:arc_length}
	Every curve $\gamma\in AC([0,1];X)$ admits an arc-length parametrization which is $1$-Lipschitz on $[0,L(\gamma)]$, where 
	$$
	L(\gamma):=\int_0^1 |\gamma'|(r)\,\mathrm{d}r.
	$$
\end{lemma}
\begin{proof}
	Define the accumulated length
	$$
	\ell(t) :=\int_0^t|\gamma'|(r)\,\mathrm{d}r.
	$$
	By Lemma ~\ref{lem:metric_length}, $d(\gamma(s),\gamma(t))\leq\ell(t)-\ell(s)$ whenever $s\leq t$. In particular, $\gamma$ is constant on each interval on which $\ell$ is constant. If $L(\gamma)=0$, the curve is constant and the assertion is trivial. Otherwise $\ell$ is continuous and maps $[0,1]$ onto $[0,L(\gamma)]$. For $\sigma \in [0,L(\gamma)]$ let
	$$
	\tau(\sigma):=\min\{t\in[0,1] \mid \ell(t)=\sigma\}, \qquad \widetilde\gamma(\sigma):=\gamma(\tau(\sigma)).
	$$
	For $sigma_1\leq\sigma_2$, we have
	\begin{align*}
		d(\widetilde\gamma(\sigma_1),\widetilde\gamma(\sigma_2)) &\leq \int_{\tau(\sigma_1)}^{\tau(\sigma_2)}|\gamma'|(r)\,\mathrm{d}r\\
																 &=\ell(\tau(\sigma_2))-\ell(\tau(\sigma_1)) = \sigma_2-\sigma_1.
	\end{align*}
\end{proof}

The preceding lemma explains why Lipschitz curves often suffice in approximation arguments. We will retain the original time variable below, since estimates in $L^2(0,1)$ allow us to avoid a separate change back from the arc-length parametrization. %TODO Was ist damit gemeint?

	\label{lem:brenier}
	Let $\mu,\nu\in\mathscr{P}_2(\mathbb{R}^n)$ and suppose that $\mu\ll\mathcal{L}^n$. For the quadratic cost $|x-y|^2$, there exists an optimal transport map $T:\mathbb{R}^n\to \mathbb{R}^n$ such that $T_\#\mu = \nu$ and 
	$$
	W_2^2(\mu,\nu) = \int_{\mathbb{R}^n}|T(x)-x|^2\,\mathrm{d}\mu(x)
	$$
	It is unique $\mu$-almost everywhere and is the gradient of a convex function $\mu$-almost everywhere
	%TODO Wurde das nicht schon davor implizit bewiesen?
\end{lemma}

The proof uses the characterization of optimal plans by cyclical monotonicity and is not reporduced here; for a detailed walkthrough see \cite[Theorem~2.3]{AmbrosioSavare2007}. The assumption on the source measure is essential. We use the lemma only after mollifying the measures, so that the source measures possess densities.

As a preperation for our next core result we first record a useful representation of the quadratic action. It is also a direct application of the convex duality and Riesz representation results recalled in Chapter~2. We introduce a representation of the kinetic energy which allows us to pass the limit in sequences of measure curves and their associated fluxes. For a narrowly continuous curve $(\mu_t)_{t\in[0,1]}$ and a velocity field $c$ we write
$$
\mathrm{d}M(t,x):=\mathrm{d}\mu_t(x)\,\mathrm dt,\qquad \mathrm{d}E(t,x):=v_t(x)\,\mathrm dM(t,x).
$$
Here, $M$ is a positive space-time measure and $E$ is a vector-valued space-time measure. Thus, the distributional continuity equation reads
$$
\int_0^1\int_{\mathbb{R}^n}\partial_t\varphi\,\mathrm dM + \int_0^1\int_{\mathbb{R}^n}\nabla_x\varphi\cdot\mathrm dE=0\qquad \forall\varphi\in C_c^\infty((0,1)\times \mathbb{R}^n).
$$
Instead of taking a weak limit of the velocity fields directly, we take a weak-* limit of the vector measures $E$ and recover a velocity field from the limit.

\begin{lemma}
Let $M\in\mathcal{M}_+((0,1)\times \mathbb{R}^n)$ and $E\in\mathcal{M}((0,1)\times \mathbb{R}^n;\mathbb{R}^n)$ be finite Radon measures. Define the action functional by
$$
\mathcal{C}(M,E) := \sup_{b\in C_c((0,1)\times \mathbb{R}^n;\mathbb{R}^n)}\left\{\int_{\mathbb{R}^n} b\cdot \mathrm dE - \frac{1}{2}\int_0^1\int_{\mathbb{R}^n} |b|^2\,\mathrm dM\right\}.
$$
Then $\mathcal{C}(M,E)<\infty$ if and only if $E=vM$ for some $v\in L^2(M; \mathbb{R}^n)$. In this case,
$$
\mathcal{(}M,E)=\frac{1}{2}\int_0^1\int_{\mathbb{R}^n} |v|^2\,\mathrm dM.
$$
Moreover, $\mathcal{C}$ is lower semicontinuous under narrow convergence of $M$ and weak-* convergence of $E$. The same assertions hold with $(0,1)\times \mathbb{R}^n$ being replaced by an open subset.

We now establish the first coneection between the metric and the dynamic perspectives. Given an absolutely continuous curve in the Wasserstein space, we construct a velocity field which generates this curve through the continuity equation. The construction consists of approximating the curve by piecewise displacement interpolations and passing to the limit. 

\begin{theorem}
	Let $(\mu_t)_{t\in[0,1]}\in AC^2([0,1];\mathscr{P}_2(\mathbb R^n))$. Then there exists a Borel velocity field $v:(0,1)\times \mathbb{R}^n\to \mathbb{R}^n$ satisfying 
	\begin{itemize}
		\item the continuity equation in the distributional sense
		\item $\lVert v_t\rVert_{L^2(\mu_t)}\leq|\mu'|(t)$ for a.e. $t\in[0,1]$.
	\end{itemize}
\end{theorem}
This theorem can also be proven for $\lVert v_t\rVert_{L^p}$, where $p>1$ (see \cite{AmbrosioSavare2007}), but we want to concern ourselves with the quadratic case. The idea is to construct a sequence of $\mu_t$ and a corresponding sequence for $v_t$ such that the continuity equation holds, $\mu_t^k$ is absolutely continuous for all $k$ and the limit admits the desired inequality.
\begin{proof}
	Let $(\eta_k)_{k\in \mathbb{N}}$ be a sequence of mollifiers with support in $B_{1/k}(0)$. We get 
	$$
	\widetilde\mu_t^k:=\mu_t*\eta_k.
	$$
	The regularized measures are absolutely continuous with respect to the Lebesgue measure, so that Lemma ~\ref{lem:brenier} applies. We distinguish $\widetilde\mu_t^k$ from the piecewise interpolated measures $\mu_t^k$ constructed below.

	Two useful consequences of convolution (with mollifiers) are 
	\begin{align*}
		W_2(\widetilde\mu_t^k,\mu_t &\leq \frac{1}{k}\\%TODO wofür
		W_2(\widetilde\mu_s^k,\widetilde\mu_t^k)&\leq W_2(\mu_s,\mu_t).
	\end{align*}
We discretize $[0,1]$ by $t_i:= i/k$, $i=0,\dots,k$. For each $i=0,\dots,k-1$, let $T^{i,k}$ be the optimal transport map from $\widetilde\mu_{t_i}^k$ to $\widetilde\mu_{t_{i+1}}^k$. Thus,
$$
(T^{i,k})_\#\widetilde\mu_{t_i}^k=\widetilde\mu_{t_{i+1}}^k,\qquad \int_{\mathbb{R}^n}|T^{i,k}(x)-x|^2\,\mathrm d\widetilde\mu_{t_i}^k(x) = W_2^2(\widetilde\mu_{t_i}^k,\widetilde\mu_{t_{i+1}}^k).
$$
For $t\in[t_i,t_{i+1}]$, define
\begin{align*}
	T_t^{i,k}(x) &:= (i+1-kt)x+(kt-i)T^{i,k}(x),\\
	\mu_t^k:=(T_t^{i,k})_\#\widetilde\mu_{t_i}^k.
\end{align*}
The coefficients interpolate linearly between $0$ and $1$. In particular, $T_{t_i}^{i,k}=\operatorname{Id}$ and $T_{t_{i+1}}^{i,k}=T^{i,k}$, so that $\mu_{t_i}^k = \widetilde\mu_{t_i}^k$ and $\mu_{t_{i+1}}^k=\widetilde\mu_{t_{i+1}}^k$. The interpolations on adjacent intervals therefore agree at their endpoints.

We differentiate the trajectories with respect to time to obtain the Lagrangian velocity
$$
v^{i,k}(x):=\frac{\mathrm d}{\mathrm dt} T_t^{i,k}(x) = k(T^{i,k}(x)-x).
$$
This velocity again is expressed in terms of the starting position $x$. For the continuity equation, we define the Eulerian velocity by 
$$
v_t^k(y) := v^{i,k}\left((T_t^{i,k})^{-1}(y)\right),\qquad t\in (t_i,t_{i+1}).
$$

Let $\varphi\in C_c^\infty((0,1)\times \mathbb{R}^n$. On each interval, the pushforward identity and the chain rule yield
\begin{align*}
	\frac{\mathrm d}{\mathrm dt}\int_{\mathbb{R}^n}\varphi(t,y)\,\mathrm d\mu_t^k(y) &= \frac{\mathrm d}{\mathrm dt}\int_{\mathbb{R}^n}\varphi(t,T_t^{i,k}(x))\,\mathrm d\widetilde\mu_{t_i}^k(x)\\
																						   &=\int_{\mathbb{R}^n}\left(\partial_t\varphi(t,T_t^{i,k}(x)) + \nabla_x\varphi(t,T_t^{i,k}(x))\cdot v^{i,k}(x)\right)\,\mathrm d\widetilde\mu_{t_i}^k(x)\\
																						   &=\int_{\mathbb{R}^n}\left(\partial_t\varphi(t,y) + \nabla_x\varphi(t,y)\cdot v_t^k(y)\right)\,\mathrm d\mu_t^k(y).
\end{align*}
Since $\varphi$ is compactly supported, the derivatives are bounded and the endpoints vanish for every derivative and since $v^{i,k}$ is integrable, differentiation under the integral is well-defined. Thus,
$$
\int_0^1\int_{\mathbb{R}^n}(\partial_t\phi+\nabla_x\phi\cdot v_t^k)\,\mathrm d\mu_t^k\,\mathrm dt=0.
$$
For the upper bound of the velocity, we use for $t\in(t_i,t_{i+1})$ the optimality of the transport map $T^{i,k}$, \eqref{moll-con} and Lemma~\ref{lem:metric-length} to obtain
\begin{align*}
	\lVert v_t^k\rVert_{L^2(\mu_t^k)}^2
	&= \int_{\mathbb{R}^n}|v^{i,k}(x)|^2\,\mathrm d\widetilde\mu_t{t_i}^k(x)\\
	&=k^2W_2^2(\widetilde\mu_{t_i}^k,\widetilde\mu_{t_{i+1}}^k\\
	&\leq k^2W_2^2(\mu_{t_i},\mu_{t_{i+1}})\\
	&\leq k^2\left(\int_{t_i}^{t_{i+1}}|\mu'|(s)\,\mathrm ds\right)^2\\
	&\leq k\int_{t_i}^{t_{i+1}}|\mu'|^2(s)\,\mathrm ds.
\end{align*}
The last inequality follows from Cauchy-Schwarz and the fact, that $t_{i+1}-t_i=\frac{1}{k}$. Integrating over each interval and summing yields
\begin{equation*}
	\int_0^1 \lVert v_t^k\rVert_{L^2(\mu_t^k)}^2\,\mathrm dt\leq \int_0^1|\mu'|^2(t)\,\mathrm dt.
\end{equation*}
We still need to show the uniform convergence of $\mu_t^k$ in time to $\mu_t$ in the space $W_2$. This is easily shown by a triangle-inequality argument, Lemma~\ref{lem_metric-length} and the convolution properties mentioned above, i.e.
\begin{align*}
	W_2(\mu_t^k,\mu_t) &\leq W_2(\mu_t^k,\widetilde\mu_{t_i}^k) + W_2(\widetilde\mu_{t_i}^k,\mu_{t_i})+W_2(\mu_{t_i},\mu_t)\\
					   &\leq 2\int_{t_i}^{t_{i+1}}|\mu'|(s)\,\mathrm ds+\frac{1}{k}\\
					   &\leq \frac{2}{\sqrt{k}}\left(\int_0^1|\mu'|^2(s)\,\mathrm ds\right)^{\frac{1}{2}} +\frac{1}{k}.
\end{align*}

Define $M^k$ and $E^k$ by

$$
Since each $\mu_t^k$ is a probability measure, using Cauchy-Schwarz gives us
\begin{align*}
	\lVert E^k\rVert_{\mathrm{TV}} &=\int_0^1\int_{\mathbb{R}^n}|v_t^k|\,\mathrm d\mu_t^k\mathrm dt\\
								   &\leq\int_0^1 \lVert v_t^k\rVert_{L^2(\mu_t^k)}\,\mathrm dt\\
								   &\leq \left(\int_0^1 \lVert v_t^k\rVert_{L^2(\mu_t^k)}^2\,\mathrm dt\right)^{\frac{1}{2}}\\
								   &\leq\left(\int_0^1|\mu|^2(t)\,\mathrm dt\right)^{\frac{1}{2}}.
\end{align*}
Hence $(E^k)_k$ is bounded and by Banach-Alaoglu 
and the separability of $C_0((0,1)\times \mathbb{R}^n;\mathbb{R}^n)$, there exists a subsequence and a finite vectore measure $E$ such that $E^k\rightharpoonup^* E$. Furthermore, uniform convergence in $W_2$ implies narrow convergence for every $t$. By dominated convergence, $M^k$ converges narrowly to $\mathrm dM(t,x)=\mathrm d\mu_t(x)\,\mathrm dt$. Passing the limit to the continuity equation yields
$$
\int_0^1\int_{\mathbb{R}^n}\partial_t\varphi\,\mathrm dM + \int_0^1\int_{\mathbb{R}^n} \nabla_x\varphi\cdot\mathrm dE=0 \qquad \forall\varphi \in C_c^\infty((0,1)\times \mathbb{R}^n).
$$

By Lemma~\ref{lem:quadratic-action} and \eqref{eq:global-energy} we have
\begin{align*}
	\mathcal{C}(ME) &\leq\liminf_{k\to\infty}\mathcal{C}(M^k,E^k)\\
					&=\frac{1}{2}\liminf_{k\to\infty}\int_0^1 \lVert v_t^k\rVert_{L^2(\mu_t^k)}^2\,\mathrm dt\\
					&\leq\frac{1}{2}\int_0^1 |\mu'|^2(t)\,\mathrm dt<\infty.
\end{align*}
Therefore we have the existence of a density $v\in L^2(M; \mathbb{R}^n)$ such that $E=vM$. Substituting this into the continuity equation above yields
$$
\int_0^1\int_{\mathbb{R}^n} (\partial_t\varphi+\nabla_x\varphi\cdot v_t)\,\mathrm d\mu_t\,\mathrm dt = 0.
$$
This proves that $(\mu_t,v_t)$ satisfies the continuity equation. To obtain the desired estimate for almost every $t$ can be seen by a lower semicontinuity argument on a subset of $(0,1)\times \mathbb{R}^n$ with the same argumentation. 
\end{proof}

	$$

\begin{theorem}[Benamou-Brenier Formula]
	\label{thm:bb}
	Let $\mu_0,\mu_1\in\mathscr{P}_2(\mathbb{R}^n)$. Then
	\begin{equation*}
		\frac{1}{2}W_2^2(\mu_0,\mu_1)=\min_{(\mu,v)}\frac{1}{2}\int_0^1\int_{\mathbb{R}^n}|v_t|^2\,\mathrm d\mu_t\,\mathrm dt
	\end{equation*}
	where the minimum is taken over narrowly continuous curves in $\mathscr{P}_2(\mathbb{R}^n)$ with the prescribed endpoints and Borel velocity fields satisfying the distributional continuity equation and having finite quadratic action
	%TODO Ersetze den Begriff "quadratic action" oder definiere davor
\end{theorem}
\begin{proof}
	First, let $(\mu,v)$ be any admissible pair. By Theorem~\ref{thm:ce-to-ac}, the curve is absolutely continuous in $W_2$ and its metric derivative is bounded by the norm of $v_t$. Lemma ~\ref{lem:metric-length} and Cuachy-Schwarz yied
	\begin{align*}
		W_2(\mu_0,\mu_1) &\leq\int_0^1|\mu'|(t)\,\mathrm dt\\
						 &\leq\int_0^1 \lVert v_t\rVert_{L^2(\mu_t)}\,\mathrm dt\\
						 &\leq\left(\int_0^1 \lVert v_t\rVert_{L^2(\mu_t)}^2\,\mathrm dt\right)^{\frac{1}{2}}.
	\end{align*}
	Conversely, let $\gamma\in\Gamma(\mu_0,\mu_1)$ be an optimal transport plan for the quadratic cost and define the interpolation by
	$$
	X_t(x,y):=(1-t)x+ty,\qquad \mu_t:=(X_t)_\#\gamma.
	$$
	The endpoint conditions follow from the marginal conditions of $\gamma$. For $0\leq s<t\leq 1$, the transport plan $(X_s,X_t)_\#\gamma$ couples $\mu_s$ and $\mu_t$ and consequently
	\begin{align*}
		W_2^2(\mu_s,\mu_t) &\leq\int_{\mathbb{R}^n\times \mathbb{R}^n}|X_t(x,y)-X_s(x,y)|^2\,\mathrm d\gamma(x,y)\\
						   &=(t-s)^2\int_{\mathbb{R}^n \times \mathbb{R}^n}|y-x|^2\,\mathrm d\gamma(x,y)\\
						   &=(t-s)^2W_2^2(\mu_0,\mu_1).
	\end{align*}
	In particular, we can assume the curve to by Lipschitz and by the triangle equality we get
	\begin{align*}
	W_2(\mu_0,\mu_1)&\leq W_2(\mu_0,\mu_s)+W_2(\mu_s,\mu_t)+W_2(\mu_t,\mu_1)\\&\leq sW_2(\mu_0,\mu_1) + (t-s)W_2(\mu_0,\mu_1)+(1-t)W_2(\mu_0,\mu_1) = W_2(\mu_0,\mu_1). 
\end{align*}
Hence $|\mu'|(t)= W_2(\mu_0,\mu_1)$ for a.e. $t$. By Corollary~\ref{cor:min}, there exists a velocity field $v$ satisfying the continuity equation and $\lVert v_t\rVert_{L^2(\mu_t)}=W_2(\mu_0,\mu_1)$ for a.e. $t$. For this admissible pair it follows
$$
\frac{1}{2}\int_0^1\int_{\mathbb{R}^n}|v_t|^2\,\mathrm d\mu_t\,\mathrm dt = \frac{1}{2} W_2^2(\mu_0,\mu_1).
$$
\end{proof}

\subsection{Numerical Analysis of the Saddle-Point Problem}
\label{sec:alg2}

In this section, we follow the augmented Lagrangian approach introduced by Benamou and Brenier \cite{BB2000} to solve the dynamic optimal transport problem. The main idea is to replace the nonlinear kinetic transport cost by a convex dual representation. The resulting saddle-point problem can then be treated by the ALG2 algorithm, whose iterations consist of solving a linear elliptic equation, performing a pointwise projection, and updating a Lagrange multiplier.

We define the Lagrangian of \eqref{eq:BB} by
\begin{equation*}
	\label{eq:alg2_lagrangian}
	\mathcal{L}(\varphi, \rho, J) = \int_0^1\int_\Omega\left(\frac{|J|^2}{2\rho} - \rho\varphi_t -J \cdot \nabla\varphi\right)\,\mathrm dx\,\mathrm dt + G(\varphi),
\end{equation*}
where 
$$
G(\varphi):=\int_\Omega\left(\rho_1(x)\varphi(1,x)-\rho_0(x)\varphi(0,x)\right)\,\mathrm dx.
$$
The constrained problem is formally equivalent to 
$$
\int_{(\rho,J)}\sup_{\varphi}\mathcal L(\rho,J,\varphi).
$$
Here $\varphi$ is a scalar function in space-time and acts as the Lagrange multiplier for the continuity equation.
\begin{lemma}
	We can rewrite $\frac{|J|^2}{2\rho}$ as the supremum of affine linear convex functions, namely
	$$
	\sup_{(a,b)\in K}(a\rho + b \cdot J),\quad K:=\left\{(a,b)\in \mathbb{R}\times \mathbb{R}^n\mid a + \frac{|b|^2}{2}\leq 0\right\}.
	$$
\end{lemma}

\begin{proof}
	First assume $\rho>0$. For fixed $b$ the expression is increasing in $a$, so its supremum under the constraint $a\leq-\frac{|b|^2}{2}$ is attained at $a = -\frac{|b|^2}{2}$. Hence
	\begin{align*}
		\sup_{(a,b)\in K}(a\rho + b\cdot J) &= \sup_{b\in \mathbb{R}^n}\left(-\frac{\rho}{2}|b|^2 + b\cdot J \right)\\
											&=\sup_{b\in \mathbb{R}^n}\left(\frac{|J|^2}{2\rho}-\frac{\rho}{2}\left|b-\frac J\rho\right|^2\right) = \frac{|J|^2}{2\rho}.
	\end{align*}
\end{proof}
For the remainder of the numerical derivation, introduce
$$
z:=\begin{pmatrix}
\rho\\J
\end{pmatrix}\qquad q:= \begin{pmatrix}
a\\b
\end{pmatrix}\qquad \langle z,q\rangle := \int_0^1\int_\Omega z\cdot q\,\mathrm dx\,\mathrm dt.
$$
In contrast to the original paper, we use $z$ instead of $\mu$ to avoid confusion with the probability measures considered in the preceding sections. Define the indicator functional
$$
F(q) := \begin{cases} 0,&q\in K,\\
	+\infty,&\text{otherwise}.
\end{cases}
$$
We can rewrite the Lagrangian as %TODO Das kommt noch aus dem nichts. Die Herleitung noch
$$
\inf_{(\rho,J)}\sup_\varphi \mathcal{L}(\varphi,\rho,J) = \sup_z\inf_{(\varphi,q)} F(q) + G(\varphi) + \langle z,\nabla_{t,x}\varphi-q\rangle
$$
For a parameter $r>0$, define
\begin{equation*}
	\label{eq:alg2}
	\mathcal{L}_r(\varphi,q,z) := F(q)-G(\varphi) + \langle z,D\varphi - q\rangle + \frac r2\|\nabla\varphi - q\|^2
\end{equation*}
Starting from $q^0$ and $z^0$, ALG2 alternates minimization with respect to $\varphi$ and $q$, followed by a multiplier update. We now derive these three steps.
\paragraph{Step A: update of $\varphi$.}
For fixed $q^k,z^k$, define
$$
L_k(\varphi):= -G(\varphi) + \langle z^k,\nabla\varphi - q^k\rangle + \frac r2\|\nabla\varphi - q^k\|^2.
$$
We aim to find a solution for the expression
$$
\mathrm{arg}\min_\varphi \mathcal{L}_r(\varphi,q^{n-1},z) = \mathrm{arg}\min_\varphi L_k(\varphi).
$$
For a test direction $\psi$, its first variation is
\begin{align*}
	\frac{\mathrm d}{\mathrm d\varepsilon} L_k(\varphi^{k+1}+\varepsilon\psi)\Big|_{\varepsilon = 0} &=-G(\psi) + \langle z^k,\nabla\psi\rangle+r\langle\nabla\varphi^{k+1}-q^k,\nabla\psi\rangle.
\end{align*}
Setting this expression equal to zero yields the weak problem
\begin{equation*}
	\label{eq:alg2_weak}
	r\int_0^1\int_\Omega\nabla\varphi^{k+1}\cdot \nabla\psi\,\mathrm dx\,\mathrm dt = G(\psi) + \int_0^1\int_\Omega(rq^k-z^k)\cdot\nabla\psi\,\mathrm dx\,\mathrm dt
\end{equation*}
for every admissible $\psi$.

This weak formulation is preferable for numerical purposes, since it contains the prescribed endpoint data through $G$. To identify its strong form suppose the iterates are smooth and set 
$$
A^k := z^k+r(\nabla\varphi^{k+1}-q^k) = \begin{pmatrix}
A_t^k\\ A_x^k.
\end{pmatrix}
$$
Then \eqref{eq:alg2_weak} is equivalent to 
$$
\int_0^1\int_\Omega A^k\cdot \nabla\psi\,\mathrm dx\,\mathrm dt = G(\psi).
$$
Integration by parts in space and times gives
\begin{align*}
	\int_0^1\int_\Omega A^k\cdot \nabla\psi &= -\int_0^1\int_\Omega\operatorname{div}_{t,x}A^k\,\psi+\int_\Omega\left(A_t^k(1,x)\psi(1,x)-A_t^k(0,x)\psi(0,x)\right)\,\mathrm dx\\
											&\quad + \int_0^1\int_{\partial\Omega}(A_x^k\cdot n)\psi\,\mathrm dS\,\mathrm dt.
\end{align*}
Consequently, the interier equation is 
\begin{equation*}
	-r\Delta_{t,x}\varphi^{k+1} = \operatorname{div}_{t,x}(z^k-rq^k)\qquad \text{in }(0,1)\times\Omega.
\end{equation*}
The time-boundary conditions are inhomogeneous, in the sense that
\begin{equation*}
	A_t^k(0,x) = \rho_0(x),\qquad A_t^k(1,x) = \rho_1(x).
\end{equation*}

For the no-flux setting in Section ~\ref{sec:dynamical_ot}, the spatial boundary condition is
$$
A_x^k\cdot n = 0\qquad \text{on }(0,1)\times\partial\Omega.
$$
In particular, the endpoint terms cannot simply be omitted. Benamou and Brenier also considered periodic boundary conditions in space, which replace the spatial boundary condition.

For completeness, we briefly discuss solvability. Assume $\rho_0,\rho_1\in L^2(\Omega)$, $q^k,z^k\in L^2((0,1)\times \Omega)$. Since the trace operator of functions in $H^1((0,1)\times \Omega)$ belong to $L^2(\Omega$, the right-hand side of \eqref{eq:alg2_weak} is continuous on $H^1((0,1)\times\Omega)$. Since the initial and final densities have equal mass, $G(1)=0$, and the right-hand side vanishes on constants. The bilinear form 
$$
a(\varphi,\psi) := r\int_0^1\int_\Omega\nabla\varphi\cdot \nabla\psi\,\mathrm dx\,\mathrm dt
$$
is continuous and, by the Poincar\'e inequality, coercive on $H^1((0,1)\times\Omega)$. The Lax-Milgram theorem therefore gives a unique solution modulo constants.

For numerical convenience, Benamou and Brenier additionally consider a small pertubation of the Laplace equation,
$$
-r\Delta_{t,x}\varphi^{k+1}+r\varepsilon\varphi^{k+1} = \operatorname{div}_{t,x}(z^k-rq^k),\qquad \varepsilon>0,
$$
with the corresponding boundary conditions.

\paragraph{Step B: update of $q$.}
For fixed $\varphi^{k+1}$ and $z^k$, all terms depending on $q$ in \eqref{eq:alg2_augmented_lagrangian} give
$$
q^{k+1}\in\operatorname*{arg\,\min}_q\left\{F(q)+\frac r2\left\|q-\left(\nabla\varphi^{k+1}+\frac{z^k}{r}\right)\right\|^2\right\}.
$$
Thus, with
$$
p^{k+1}:=\nabla\varphi^{k+1}+\frac{z^k}{r},
$$
the update is the pointwise projection. We solve
\begin{equation*}
	\min_{(a,b)\in K}\left((a-\alpha)^2+|b-\beta|^2\right).
\end{equation*}
If $\alpha +\frac{|beta|^2}{2}\leq0$, the point $p$ is already feasible. Otherwise the constraint is active. Introducing a scalar Lagrange multiplier $\lambda>0$, we consider
$$
\mathcal{H}(a,b,\lambda) = (a-\alpha)^2+|b-\beta|^2+\lambda\left(a+\frac{|b|^2}{2}\right).
$$
The stationarity conditions read
$$
2(a-\alpha)+\lambda=0,\qquad 2(b-\beta)+\lambda b=0,
$$
so that 
$$
a=\alpha-\frac{\lambda}{2},\qquad b=\frac{2\beta}{2+\lambda}.
$$
Substitution into the active constraint yields
\begin{equation*}
	H(\lambda) := \alpha-\frac\lambda2+\frac{2|\beta|^2}{2+\lambda)^2}=0.
\end{equation*}
Since 
$$
H'(\lambda) = -\frac12-\frac{4|\beta|^2}{(2+\lambda)^3}<0,
$$
$H$ is strictly decreasing. In the active case $H(0) = \alpha+\frac{|\beta|^2}{2}>0$, while $H(\lambda)\to-\infty$ as $\lambda\to\infty$. Hence there exists exactly one positive solution. It can be computed by using Newton's method,
$$
\lambda_{j+1}=\lambda_j-\frac{H(\lambda_j)}{H'(\lambda_j)},
$$
and in the implementation ensuring $\lambda_j\geq 0$. The projected point for $(a,b)$ is finally given by
$$
\left(\alpha-\frac\lambda2,\frac{2\beta}{2+\lambda}\right).
$$

\paragraph{Setp C: multiplier update.}
The remaining step is an ascent step for the Lagrange multiplier $z= (\rho,J)$:
\begin{equation*}
	z^{k+1}=z^k+r\left(\nabla\varphi^{k+1}-q^{k+1}\right).
\end{equation*}

\begin{algorithm}[H]
\caption{ALG2 for quadratic dynamical optimal transport}
\label{alg:bb_alg2}
\begin{algorithmic}[1]
    \Require Initial iterates $q^0,z^0$, parameter $r>0$,
             tolerance $\mathrm{tol}>0$, maximum iterations $N_{\max}$
    \For{$k=0,\ldots,N_{\max}-1$}
        \State Find $\varphi^{k+1}\in H^1((0,1)\times\Omega)$ such that
        \Statex \hspace{1em}$r\langle \nabla\varphi^{k+1},\nabla\psi\rangle
          =G(\psi)+\langle rq^k-z^k,\nabla\psi\rangle$
          for all $\psi\in H^1((0,1)\times\Omega)/$.
        \State $p^{k+1}\gets \nabla\varphi^{k+1}+z^k/r$
        \State $q^{k+1}(t,x)\gets P_K(p^{k+1}(t,x))$
               for all space-time degrees of freedom %TODO Das noch ändern
        \State $z^{k+1}\gets z^k+r(\nabla\varphi^{k+1}-q^{k+1})$
        \State $R^{k+1}\gets\|\nabla\varphi^{k+1}-q^{k+1}\|$
        \If{$R^{k+1}<\mathrm{tol}$}
            \State \textbf{break}
        \EndIf
    \EndFor
    \Ensure $\varphi^{k+1},q^{k+1},z^{k+1}$
\end{algorithmic}
\end{algorithm}
